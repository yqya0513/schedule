function renderSchedule() { const grid = document.getElementById('schedule-grid'); grid.innerHTML = ''; // 1. 時間軸のガイド線描画 for (let i = 0; i < 24; i++) { const line = document.createElement('div'); line.className = 'time-slot-line'; line.style.top = `${i * 60}px`; line.innerText = `${String(i).padStart(2, '0')}:00`; grid.appendChild(line); } // 2. 現在時刻線の描画 const now = new Date(); const currentMin = now.getHours() * 60 + now.getMinutes(); const timeLine = document.createElement('div'); timeLine.className = 'current-time-line'; timeLine.style.top = `${currentMin}px`; grid.appendChild(timeLine); // 3. 全ての予定（単発・毎日・毎月等）を統合して分単位に変換 // ※もし自動生成データがある場合はここで schedules に合流させてソートします const items = schedules.map(s => ({ ...s, startMin: timeToMinutes(s.start), endMin: timeToMinutes(s.end) })).sort((a, b) => a.startMin - b.startMin || (b.endMin - b.startMin) - (a.endMin - a.startMin)); // 4. 重なり判定とカラム（列）割り当ての計算 const clusters = []; let currentCluster = []; let clusterEnd = -1; items.forEach(item => { if (currentCluster.length === 0 || item.startMin < clusterEnd) { currentCluster.push(item); clusterEnd = Math.max(clusterEnd, item.endMin); } else { clusters.push(currentCluster); currentCluster = [item]; clusterEnd = item.endMin; } }); if (currentCluster.length > 0) { clusters.push(currentCluster); } // 5. 各クラスタ内での列の最適化配置 const gridWidth = grid.clientWidth - 45; clusters.forEach(cluster => { const columns = []; cluster.forEach(item => { let placed = false; for (let i = 0; i < columns.length; i++) { const lastInCol = columns[i][columns[i].length - 1]; if (lastInCol.endMin <= item.startMin) { columns[i].push(item); item.colIndex = i; placed = true; break; } } if (!placed) { item.colIndex = columns.length; columns.push([item]); } }); const maxCols = columns.length; cluster.forEach(item => { const el = document.createElement('div'); el.className = 'schedule-item'; const top = item.startMin; const height = Math.max(item.endMin - item.startMin, 18); const itemWidth = (gridWidth / maxCols) - 2; // 余白調整 const left = 45 + (itemWidth * item.colIndex) + (item.colIndex * 2); el.style.top = `${top}px`; el.style.height = `${height}px`; el.style.width = `${itemWidth}px`; el.style.left = `${left}px`; el.innerText = `${item.title} (${item.start}~${item.end})`; el.onclick = (e) => { e.stopPropagation(); openEditModal(item.id); }; grid.appendChild(el); }); }); }
